import { connect } from 'node:http2';
import { createPrivateKey, sign } from 'node:crypto';
import { readFileSync } from 'node:fs';
import { query } from './db.js';

let credential;
let jwt;
let jwtAt = 0;
const last = new Map();
function auth() {
  if (!credential) credential = JSON.parse(readFileSync(process.env.APNS_CREDENTIALS_PATH || '/opt/noda/credentials/apns.json', 'utf8'));
  const now = Math.floor(Date.now() / 1000);
  if (!jwt || now - jwtAt > 3000) {
    const encode = value => Buffer.from(JSON.stringify(value)).toString('base64url');
    const body = `${encode({ alg: 'ES256', kid: credential.keyId })}.${encode({ iss: credential.teamId, iat: now })}`;
    jwt = `${body}.${sign('sha256', Buffer.from(body), { key: createPrivateKey(credential.key), dsaEncoding: 'ieee-p1363' }).toString('base64url')}`;
    jwtAt = now;
  }
  return jwt;
}
export function activityPayload(agent, now = Date.now()) {
  const running = agent?.state === 'working';
  const props = { key: agent?.key || '', project: agent?.project || 'Arra', since: agent?.since || now,
    running, status: running ? 'Агент работает' : agent?.state === 'error' ? 'Нужна помощь' : 'Ждет тебя' };
  return { aps: { timestamp: Math.floor(now / 1000), event: running ? 'update' : 'end',
    'content-state': { name: 'ArraAgentActivity', props: JSON.stringify(props) },
    ...(running ? { 'stale-date': Math.floor(now / 1000) + 180 } : { 'dismissal-date': Math.floor(now / 1000) + 900 }) } };
}
export function sendActivityPush(token, payload) {
  const authorization = `bearer ${auth()}`;
  return new Promise((resolve, reject) => {
    const client = connect('https://api.push.apple.com');
    client.on('error', reject);
    const req = client.request({ ':method': 'POST', ':path': `/3/device/${token}`, authorization,
      'apns-push-type': 'liveactivity', 'apns-topic': 'com.arratima.aura.push-type.liveactivity', 'apns-priority': '5' });
    let status = 0;
    req.setTimeout(10000, () => { req.close(); client.close(); reject(new Error('APNs timeout')); });
    req.on('response', headers => { status = Number(headers[':status']); });
    req.on('data', () => {});
    req.on('error', reject);
    req.on('end', () => { client.close(); resolve(status); });
    req.end(JSON.stringify(payload));
  });
}
export async function updateAgentActivities(userId, agents) {
  const { rows } = await query('SELECT token, agent_key FROM ara_live_activities WHERE user_id=$1 AND expires_at > now()', [userId]);
  for (const row of rows) {
    const agent = agents.find(a => a.key === row.agent_key);
    // Offline computers are not evidence that the task has finished.
    if (!agent) continue;
    const previous = last.get(row.token);
    const fingerprint = JSON.stringify([agent.state, agent.project, agent.since]);
    if (previous?.fingerprint === fingerprint && Date.now() - previous.at < 60000) continue;
    try {
      const status = await sendActivityPush(row.token, activityPayload(agent));
      if (status === 200) last.set(row.token, { fingerprint, at: Date.now() });
      if (status === 410 || status === 400 || (status === 200 && agent.state !== 'working')) {
        await query('DELETE FROM ara_live_activities WHERE token=$1', [row.token]); last.delete(row.token);
      }
    } catch (error) { console.warn('Live Activity update failed:', error.message); }
  }
}
